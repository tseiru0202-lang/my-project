  <script>
    'use strict';

    const API_URL = 'https://www.googleapis.com/youtube/v3/search';
    const STORAGE_KEY = 'clipbox_youtube_api_key_v1';
    const state = {
      apiKey: localStorage.getItem(STORAGE_KEY) || '',
      nextPageToken: '',
      currentQuery: '',
      currentVideo: null,
      loading: false,
      cache: new Map()
    };

    const $ = (id) => document.getElementById(id);
    const statusEl = $('status');
    const keyNotice = $('keyNotice');

    function setStatus(message, kind = '') {
      statusEl.textContent = message;
      statusEl.className = `status ${kind}`.trim();
    }

    function updateKeyNotice() {
      if (state.apiKey) {
        keyNotice.textContent = 'APIキーがこの端末のブラウザに保存されています。検索を始められます。';
        keyNotice.style.borderColor = 'rgba(110,210,150,.35)';
        keyNotice.style.background = 'rgba(50,130,85,.12)';
      } else {
        keyNotice.textContent = '最初に「APIキー設定」からYouTube Data API v3のキーを登録してください。';
        keyNotice.style.borderColor = '';
        keyNotice.style.background = '';
      }
    }

    function setLoading(isLoading) {
      state.loading = isLoading;
      $('searchButton').disabled = isLoading;
      $('loadMoreButton').disabled = isLoading;
      $('searchButton').textContent = isLoading ? '検索中…' : '検索する';
      $('loadMoreButton').textContent = isLoading ? '読み込み中…' : 'さらに読み込む';
    }

    function showEmpty(message, container = $('resultsGrid')) {
      container.replaceChildren();
      const box = document.createElement('div');
      box.className = 'empty';
      box.textContent = message;
      container.appendChild(box);
    }

    function makeThumbnail(item) {
      const thumbnails = item.snippet?.thumbnails || {};
      return thumbnails.medium?.url || thumbnails.high?.url || thumbnails.default?.url || '';
    }

    function createVideoCard(item) {
      const videoId = item.id?.videoId;
      if (!videoId || !item.snippet) return null;
      const button = document.createElement('button');
      button.type = 'button';
      button.className = 'video-card';
      button.setAttribute('aria-label', `${item.snippet.title}を再生`);

      const image = document.createElement('img');
      image.src = makeThumbnail(item);
      image.alt = '';
      image.loading = 'lazy';
      image.referrerPolicy = 'strict-origin-when-cross-origin';

      const info = document.createElement('div');
      info.className = 'video-info';
      const title = document.createElement('strong');
      title.textContent = item.snippet.title || 'タイトル不明';
      const channel = document.createElement('span');
      channel.textContent = item.snippet.channelTitle || 'チャンネル不明';
      info.append(title, channel);
      button.append(image, info);
      button.addEventListener('click', () => selectVideo(item));
      return button;
    }

    function createRelatedItem(item) {
      const videoId = item.id?.videoId;
      if (!videoId || !item.snippet) return null;
      const button = document.createElement('button');
      button.type = 'button';
      button.className = 'related-item';
      const image = document.createElement('img');
      image.src = makeThumbnail(item);
      image.alt = '';
      image.loading = 'lazy';
      const text = document.createElement('div');
      const title = document.createElement('strong');
      title.textContent = item.snippet.title || 'タイトル不明';
      const channel = document.createElement('span');
      channel.textContent = item.snippet.channelTitle || 'チャンネル不明';
      text.append(title, channel);
      button.append(image, text);
      button.addEventListener('click', () => selectVideo(item));
      return button;
    }

    function renderResults(items, append = false) {
      const grid = $('resultsGrid');
      if (!append) grid.replaceChildren();
      let added = 0;
      for (const item of items || []) {
        const card = createVideoCard(item);
        if (card) { grid.appendChild(card); added += 1; }
      }
      if (!append && added === 0) showEmpty('動画が見つかりませんでした。別のキーワードで検索してみてください。');
      const total = grid.querySelectorAll('.video-card').length;
      $('resultCount').textContent = total ? `${total}件表示中` : '';
    }

    function parseApiError(data, response) {
      const reason = data?.error?.errors?.[0]?.reason || '';
      if (reason === 'keyInvalid' || response.status === 400) return 'APIキーが正しくない可能性があります。「APIキー設定」から確認してください。';
      if (reason === 'quotaExceeded' || reason === 'dailyLimitExceeded') return 'YouTube APIの1日の検索上限に達したようです。上限が戻ってから試してください。';
      if (reason === 'accessNotConfigured') return 'Google Cloudで「YouTube Data API v3」が有効になっていません。APIを有効にしてください。';
      if (response.status === 403) return 'APIの利用が許可されませんでした。APIキーの制限設定や利用上限を確認してください。';
      return data?.error?.message || `検索に失敗しました（HTTP ${response.status}）。`;
    }

    async function searchYouTube(query, pageToken = '') {
      if (!state.apiKey) {
        $('settingsPanel').classList.add('open');
        setStatus('先にYouTube APIキーを登録してください。', 'error');
        return null;
      }
      const cacheKey = `${query}::${pageToken}`;
      if (state.cache.has(cacheKey)) return state.cache.get(cacheKey);

      const params = new URLSearchParams({
        part: 'snippet',
        type: 'video',
        q: query,
        maxResults: '12',
        videoEmbeddable: 'true',
        safeSearch: 'moderate',
        regionCode: 'JP',
        relevanceLanguage: 'ja',
        key: state.apiKey
      });
      if (pageToken) params.set('pageToken', pageToken);

      const response = await fetch(`${API_URL}?${params.toString()}`);
      const data = await response.json().catch(() => ({}));
      if (!response.ok || data.error) throw new Error(parseApiError(data, response));
      const result = { items: data.items || [], nextPageToken: data.nextPageToken || '' };
      state.cache.set(cacheKey, result);
      return result;
    }

    async function runSearch(query, pageToken = '', append = false) {
      const cleanQuery = query.trim();
      if (!cleanQuery) { setStatus('検索キーワードを入力してください。', 'error'); return; }
      if (state.loading) return;
      if (!append) {
        state.currentQuery = cleanQuery;
        state.nextPageToken = '';
        $('resultsHeading').textContent = `「${cleanQuery}」の検索結果`;
        showEmpty('検索しています…');
      }
      setLoading(true);
      setStatus('YouTubeから動画を検索しています…');
      try {
        const result = await searchYouTube(cleanQuery, pageToken);
        if (!result) return;
        renderResults(result.items, append);
        state.nextPageToken = result.nextPageToken;
        $('loadMoreButton').hidden = !state.nextPageToken;
        setStatus('検索が完了しました。動画を選ぶとこのページ内で再生できます。', 'good');
      } catch (error) {
        if (!append) showEmpty('検索結果を表示できませんでした。');
        setStatus(error.message || '予期しないエラーが発生しました。', 'error');
        $('loadMoreButton').hidden = true;
      } finally {
        setLoading(false);
      }
    }

    function relatedQueryFor(video) {
      // YouTube公式の関連動画一覧ではなく、タイトルとチャンネル名による再検索です。
      const title = (video.snippet?.title || '').replace(/[【】\[\]()（）「」『』]/g, ' ').replace(/\s+/g, ' ').trim();
      const channel = (video.snippet?.channelTitle || '').trim();
      const query = `${title} ${channel}`.trim();
      return query.slice(0, 90);
    }

    async function loadRelated(video) {
      const list = $('relatedList');
      showEmpty('関連キーワードで検索しています…', list);
      $('relatedDescription').textContent = '動画のタイトル・チャンネル名を使って、似たテーマの動画を探しています。';
      try {
        const query = relatedQueryFor(video);
        const result = await searchYouTube(query);
        if (!result) return;
        list.replaceChildren();
        const related = result.items.filter(item => item.id?.videoId !== video.id?.videoId).slice(0, 6);
        for (const item of related) {
          const element = createRelatedItem(item);
          if (element) list.appendChild(element);
        }
        if (!list.children.length) showEmpty('関連する動画が見つかりませんでした。', list);
        $('relatedDescription').textContent = 'タイトル・チャンネル名で再検索した候補です（YouTube公式の個人向けおすすめとは異なります）。';
      } catch (error) {
        showEmpty('関連動画を読み込めませんでした。', list);
        $('relatedDescription').textContent = error.message;
      }
    }

    function selectVideo(item) {
      const videoId = item.id?.videoId;
      const snippet = item.snippet;
      if (!videoId || !snippet) return;
      state.currentVideo = item;
      const frame = $('playerFrame');
      frame.replaceChildren();
      const iframe = document.createElement('iframe');
      iframe.src = `https://www.youtube-nocookie.com/embed/${encodeURIComponent(videoId)}?playsinline=1&rel=0`;
      iframe.title = snippet.title || 'YouTube video player';
      iframe.allow = 'accelerometer; autoplay; clipboard-write; encrypted-media; gyroscope; picture-in-picture; web-share';
      iframe.referrerPolicy = 'strict-origin-when-cross-origin';
      iframe.allowFullscreen = true;
      frame.appendChild(iframe);

      $('playingTitle').textContent = snippet.title || 'タイトル不明';
      $('playingChannel').textContent = snippet.channelTitle || 'チャンネル不明';
      const links = $('playerLinks');
      links.replaceChildren();
      const openLink = document.createElement('a');
      openLink.href = `https://www.youtube.com/watch?v=${encodeURIComponent(videoId)}`;
      openLink.target = '_blank';
      openLink.rel = 'noopener noreferrer';
      openLink.textContent = 'YouTubeで開く ↗';
      links.appendChild(openLink);
      links.hidden = false;
      loadRelated(item);
      setStatus('動画を選択しました。再生できない動画はYouTube側の設定で埋め込みが制限されている場合があります。', 'good');
      if (window.matchMedia('(max-width: 760px)').matches) {
        $('playerFrame').scrollIntoView({ behavior: 'smooth', block: 'start' });
      }
    }

    $('settingsToggle').addEventListener('click', () => {
      $('settingsPanel').classList.toggle('open');
      if (state.apiKey) $('apiKeyInput').value = state.apiKey;
    });

    $('saveKeyButton').addEventListener('click', () => {
      const key = $('apiKeyInput').value.trim();
      if (!key) { setStatus('APIキーを入力してください。', 'error'); return; }
      state.apiKey = key;
      localStorage.setItem(STORAGE_KEY, key);
      state.cache.clear();
      updateKeyNotice();
      setStatus('APIキーをこの端末に保存しました。検索できます。', 'good');
      $('settingsPanel').classList.remove('open');
    });

    $('removeKeyButton').addEventListener('click', () => {
      state.apiKey = '';
      localStorage.removeItem(STORAGE_KEY);
      $('apiKeyInput').value = '';
      state.cache.clear();
      updateKeyNotice();
      setStatus('保存していたAPIキーを削除しました。', 'good');
    });

    $('searchForm').addEventListener('submit', (event) => {
      event.preventDefault();
      runSearch($('searchInput').value);
    });

    $('loadMoreButton').addEventListener('click', () => {
      if (state.nextPageToken) runSearch(state.currentQuery, state.nextPageToken, true);
    });

    updateKeyNotice();
    if (state.apiKey) $('apiKeyInput').value = state.apiKey;
  </script>
